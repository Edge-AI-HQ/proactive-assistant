```python
# backend/src/pinai_agents/web/services/schedule_spec.py
"""结构化 schedule 意图 → pai_scheduled_tasks 触发字段。

纯函数、无 IO:agent 解析语言(传 type=once/interval/cron 的结构),
这里负责时间数学(相对时间用 server 的钟、本地时间按 task timezone
换算 UTC)、cron 表达式拼装(agent 不直写 cron)、展示 label 和
相邻触发 gap(供 service 层套 per-mode 最小间隔护栏)。

spec 形状(snake_case,route 层已归一):
  {"type": "once",     "in_minutes": 5}
  {"type": "once",     "at_local": "2026-06-10T15:00"}
  {"type": "interval", "every_seconds": 300}
  {"type": "cron",     "frequency": "daily",  "local_time": "09:00"}
  {"type": "cron",     "frequency": "weekly", "local_time": "09:00",
                       "days_of_week": [1, 7]}   # ISO: 1=周一 … 7=周日
"""

from __future__ import annotations

from dataclasses import dataclass
from datetime import UTC, datetime, timedelta
from typing import Any, Literal
from zoneinfo import ZoneInfo

from croniter import croniter

_WEEKDAY_ZH = {1: "一", 2: "二", 3: "三", 4: "四", 5: "五", 6: "六", 7: "日"}
_WEEKDAY_EN = {1: "Mon", 2: "Tue", 3: "Wed", 4: "Thu", 5: "Fri", 6: "Sat", 7: "Sun"}


class ScheduleSpecError(Exception):
    """非法 schedule 结构。service 层转译成 ScheduledGeneratorValidationError。"""


@dataclass(slots=True)
class ResolvedSchedule:
    trigger_kind: Literal["once", "schedule"]
    cron_expr: str | None
    interval_seconds: int | None
    next_run_at: datetime  # 首次触发(UTC aware)
    gap_seconds: int | None  # 相邻两次触发间隔;once 为 None
    schedule_key: str  # 语义签名组件,如 "interval:300" / "cron:daily:09:00"
    schedule_label: str  # 人类可读,创建时固化进 payload


def resolve_schedule(
    spec: dict[str, Any],
    *,
    timezone: str,
    language: str = "zh",
    now: datetime | None = None,
) -> ResolvedSchedule:
    now = now or datetime.now(UTC)
    if now.tzinfo is None:
        now = now.replace(tzinfo=UTC)
    try:
        tz = ZoneInfo(timezone)
    except Exception as e:
        raise ScheduleSpecError(f"invalid timezone: {timezone}") from e

    spec_type = str(spec.get("type") or "")
    if spec_type == "once":
        return _resolve_once(spec, tz=tz, language=language, now=now)
    if spec_type == "interval":
        return _resolve_interval(spec, language=language, now=now)
    if spec_type == "cron":
        return _resolve_cron(spec, tz=tz, timezone=timezone, language=language, now=now)
    raise ScheduleSpecError(f"schedule.type must be once|interval|cron, got {spec_type!r}")


def _resolve_once(
    spec: dict[str, Any], *, tz: ZoneInfo, language: str, now: datetime
) -> ResolvedSchedule:
    in_minutes = spec.get("in_minutes")
    at_local = spec.get("at_local")
    if (in_minutes is None) == (at_local is None):
        raise ScheduleSpecError("once schedule needs exactly one of in_minutes / at_local")
    if in_minutes is not None:
        minutes = int(in_minutes)
        if minutes < 1:
            raise ScheduleSpecError("in_minutes must be >= 1")
        run_at = now + timedelta(minutes=minutes)
    else:
        try:
            naive = datetime.strptime(str(at_local), "%Y-%m-%dT%H:%M")
        except ValueError as e:
            raise ScheduleSpecError(f"at_local must be YYYY-MM-DDTHH:MM, got {at_local!r}") from e
        run_at = naive.replace(tzinfo=tz).astimezone(UTC)
        if run_at <= now:
            raise ScheduleSpecError("once schedule time is in the past")
    local = run_at.astimezone(tz)
    label = (
        f"一次 · {local.strftime('%m-%d %H:%M')}"
        if language == "zh"
        else f"Once · {local.strftime('%m-%d %H:%M')}"
    )
    return ResolvedSchedule(
        trigger_kind="once",
        cron_expr=None,
        interval_seconds=None,
        next_run_at=run_at,
        gap_seconds=None,
        schedule_key=f"once:{run_at.strftime('%Y-%m-%dT%H:%MZ')}",
        schedule_label=label,
    )


def _resolve_interval(spec: dict[str, Any], *, language: str, now: datetime) -> ResolvedSchedule:
    raw = spec.get("every_seconds")
    seconds = int(raw) if raw is not None else 0
    if seconds < 1:
        raise ScheduleSpecError("every_seconds must be >= 1")
    if language == "zh":
        label = (
            f"每 {seconds // 3600} 小时"
            if seconds % 3600 == 0
            else f"每 {max(seconds // 60, 1)} 分钟"
        )
    else:
        label = (
            f"Every {seconds // 3600} hours"
            if seconds % 3600 == 0
            else f"Every {max(seconds // 60, 1)} min"
        )
    return ResolvedSchedule(
        trigger_kind="schedule",
        cron_expr=None,
        interval_seconds=seconds,
        next_run_at=now + timedelta(seconds=seconds),
        gap_seconds=seconds,
        schedule_key=f"interval:{seconds}",
        schedule_label=label,
    )


def _resolve_cron(
    spec: dict[str, Any], *, tz: ZoneInfo, timezone: str, language: str, now: datetime
) -> ResolvedSchedule:
    frequency = str(spec.get("frequency") or "")
    if frequency not in ("daily", "weekly"):
        raise ScheduleSpecError("cron schedule.frequency must be daily|weekly")
    hour, minute = _parse_local_time(str(spec.get("local_time") or ""))
    if frequency == "daily":
        cron_expr = f"{minute} {hour} * * *"
        key = f"cron:daily:{hour:02d}:{minute:02d}"
        label = (
            f"每天 {hour:02d}:{minute:02d}" if language == "zh" else f"Daily {hour:02d}:{minute:02d}"
        )
    else:
        raw_days = spec.get("days_of_week")
        if not isinstance(raw_days, list) or not raw_days:
            raise ScheduleSpecError("weekly schedule needs days_of_week (ISO 1=Mon … 7=Sun)")
        iso_days = sorted({int(d) for d in raw_days})
        if any(d < 1 or d > 7 for d in iso_days):
            raise ScheduleSpecError("days_of_week values must be 1..7 (ISO)")
        cron_days = sorted({0 if d == 7 else d for d in iso_days})  # cron: 0=Sun
        cron_expr = f"{minute} {hour} * * {','.join(str(d) for d in cron_days)}"
        key = f"cron:weekly:{','.join(str(d) for d in iso_days)}:{hour:02d}:{minute:02d}"
        if language == "zh":
            label = f"每周{'/'.join(_WEEKDAY_ZH[d] for d in iso_days)} {hour:02d}:{minute:02d}"
        else:
            label = (
                f"Weekly {'/'.join(_WEEKDAY_EN[d] for d in iso_days)} {hour:02d}:{minute:02d}"
            )
    first, second = _next_two_fires(cron_expr, tz=tz, now=now)
    return ResolvedSchedule(
        trigger_kind="schedule",
        cron_expr=cron_expr,
        interval_seconds=None,
        next_run_at=first,
        gap_seconds=int((second - first).total_seconds()),
        schedule_key=key,
        schedule_label=label,
    )


def _parse_local_time(local_time: str) -> tuple[int, int]:
    try:
        hour_raw, minute_raw = local_time.split(":", 1)
        hour, minute = int(hour_raw), int(minute_raw)
    except ValueError as e:
        raise ScheduleSpecError(f"local_time must be HH:MM, got {local_time!r}") from e
    if not (0 <= hour <= 23 and 0 <= minute <= 59):
        raise ScheduleSpecError(f"local_time out of range: {local_time!r}")
    return hour, minute


def _next_two_fires(cron_expr: str, *, tz: ZoneInfo, now: datetime) -> tuple[datetime, datetime]:
    it = croniter(cron_expr, now.astimezone(tz))
    first = it.get_next(datetime).astimezone(UTC)
    second = it.get_next(datetime).astimezone(UTC)
    return first, second
```

```python
    async def list_by_kinds(
        self,
        *,
        kinds: Sequence[str],
        include_archived: bool = False,
    ) -> list[ScheduledTaskRow]:
        async with self._conn.cursor() as cur:
            await cur.execute(
                """
                SELECT * FROM pai_scheduled_tasks
                WHERE user_id = %s
                  AND kind = ANY(%s)
                  AND (%s OR status <> 'archived')
                ORDER BY created_at DESC
                """,
                (self._user_id, list(kinds), include_archived),
            )
            rows = await cur.fetchall()
        return [ScheduledTaskRow.from_record(r) for r in rows]

    async def count_by_kinds(self, *, kinds: Sequence[str]) -> int:
        """active+paused 任务数(护栏用);archived 不计。"""
        async with self._conn.cursor() as cur:
            await cur.execute(
                """
                SELECT COUNT(*) AS n FROM pai_scheduled_tasks
                WHERE user_id = %s AND kind = ANY(%s) AND status <> 'archived'
                """,
                (self._user_id, list(kinds)),
            )
            row = await cur.fetchone()
        return int(row["n"]) if row else 0

    async def get_by_id_and_kinds(
        self, *, task_id: str, kinds: Sequence[str]
    ) -> ScheduledTaskRow | None:
        async with self._conn.cursor() as cur:
            await cur.execute(
                """
                SELECT * FROM pai_scheduled_tasks
                WHERE user_id = %s AND id = %s AND kind = ANY(%s)
                LIMIT 1
                """,
                (self._user_id, task_id, list(kinds)),
            )
            row = await cur.fetchone()
        return ScheduledTaskRow.from_record(row) if row is not None else None

    async def pause_by_kinds(
        self, *, task_id: str, kinds: Sequence[str]
    ) -> ScheduledTaskRow | None:
        async with self._conn.cursor() as cur:
            await cur.execute(
                """
                UPDATE pai_scheduled_tasks
                SET status = 'paused', updated_at = NOW()
                WHERE user_id = %s AND id = %s AND kind = ANY(%s) AND status = 'active'
                RETURNING *
                """,
                (self._user_id, task_id, list(kinds)),
            )
            row = await cur.fetchone()
        return ScheduledTaskRow.from_record(row) if row is not None else None

    async def resume_by_kinds(
        self,
        *,
        task_id: str,
        kinds: Sequence[str],
        next_run_at: datetime | None,
    ) -> ScheduledTaskRow | None:
        async with self._conn.cursor() as cur:
            await cur.execute(
                """
                UPDATE pai_scheduled_tasks
                SET status = 'active', next_run_at = %s, updated_at = NOW()
                WHERE user_id = %s AND id = %s AND kind = ANY(%s) AND status = 'paused'
                RETURNING *
                """,
                (next_run_at, self._user_id, task_id, list(kinds)),
            )
            row = await cur.fetchone()
        return ScheduledTaskRow.from_record(row) if row is not None else None

    async def archive_by_kinds(
        self, *, task_id: str, kinds: Sequence[str]
    ) -> ScheduledTaskRow | None:
        async with self._conn.cursor() as cur:
            await cur.execute(
                """
                UPDATE pai_scheduled_tasks
                SET status = 'archived', next_run_at = NULL, updated_at = NOW()
                WHERE user_id = %s
```

```python
def _language_for(original_request: str) -> str:
    return "zh" if _output_language_for(original_request) == "zh-Hans" else "en"


def _validate_reminder(mode: str, reminder: dict[str, Any] | None) -> dict[str, str]:
    if mode != "reminder":
        return {}
    reminder = _normalize_payload_keys(reminder or {})
    title = str(reminder.get("title") or "").strip()
    message = str(reminder.get("message") or "").strip()
    if not title or not message:
        raise ScheduledGeneratorValidationError("reminder mode needs reminder.title and reminder.message")
    if len(title) > 80 or len(message) > 500:
        raise ScheduledGeneratorValidationError("reminder.title <= 80 chars, reminder.message <= 500 chars")
    return {"title": title, "message": message}


def _resolve_and_guard(
    *,
    schedule: dict[str, Any],
    timezone: str,
    mode: str,
    original_request: str,
) -> ResolvedSchedule:
    if mode not in _MODE_TO_KIND:
        raise ScheduledGeneratorValidationError(f"mode must be reminder|generate, got {mode!r}")
    settings = get_web_settings()
    try:
        resolved = resolve_schedule(
            _normalize_payload_keys(schedule),
            timezone=timezone,
            language=_language_for(original_request),
        )
    except ScheduleSpecError as e:
        raise ScheduledGeneratorValidationError(str(e)) from e
    if resolved.trigger_kind == "once":
        max_ahead = timedelta(days=settings.scheduled_once_max_days_ahead)
        if resolved.next_run_at > datetime.now(UTC) + max_ahead:
            raise ScheduledGeneratorValidationError(
                f"once schedule must be within {settings.scheduled_once_max_days_ahead} days"
            )
    elif resolved.gap_seconds is not None:
        min_gap = (
            settings.scheduled_reminder_min_gap_seconds
            if mode == "reminder"
            else settings.scheduled_generate_min_gap_seconds
        )
        if resolved.gap_seconds < min_gap:
            raise ScheduledGeneratorValidationError(
                f"{mode} tasks fire at most every {min_gap} seconds (got {resolved.gap_seconds})"
            )
    return resolved


async def _guard_max_active(repo: ScheduledTasksRepository) -> None:
```

```python
def _find_similar(
    tasks: list[ScheduledTaskRow],
    *,
    semantic_signature: str,
    parsed_intent: dict[str, Any],
    confirmed_brief: dict[str, Any],
) -> list[ScheduledTaskRow]:
    schedule = parsed_intent.get("schedule")
    target_schedule = schedule if isinstance(schedule, dict) else {}
    target_topic = _topic_key(
        str(confirmed_brief.get("topic") or parsed_intent.get("topic") or "")
    )
    matches: list[ScheduledTaskRow] = []
    for row in tasks:
        payload = row.payload or {}
        if str(payload.get("semantic_signature") or "") == semantic_signature:
            matches.append(row)
            continue
        # PR65 存量行 fallback:frequency/local_time/timezone/topic 四字段精确匹配
        raw_intent = payload.get("parsed_intent")
        candidate = raw_intent if isinstance(raw_intent, dict) else {}
        legacy_match = (
            str(candidate.get("frequency") or "") == "daily"
            and target_schedule.get("type") == "cron"
            and target_schedule.get("frequency") == "daily"
            and str(candidate.get("local_time") or "") == str(target_schedule.get("local_time") or "")
            and str(candidate.get("timezone") or row.timezone or "")
            == str(parsed_intent.get("timezone") or "")
            and _topic_key(str(candidate.get("topic") or "")) == target_topic
        )
        if legacy_match:
            matches.append(row)
    return matches
```

```python
def _build_reminder_payload(*, title: str, message: str, task_id: str) -> PushPayload:
    # 同一任务的重复提醒折叠成一条 banner;不同任务互不折叠。
    return PushPayload(
        title=title[:80],
        body=message[:240],
        deep_link="app://cards",
        collapse_key=f"reminder_{task_id}",
    )


async def notify_reminder(
    pool,
    *,
    user_id: int,
    title: str,
    message: str,
    task_id: str,
) -> int:
    """用户显式创建的提醒 — 不受 notify_new_cards 开关约束(那是系统
    广播的 opt-in;这是用户自己要求的打扰),与 notify_proactive_message
    同一豁免逻辑。卡片已落库,push 失败不重试任务,所以永不上抛。"""
    if not message.strip():
        return 0
    try:
        async with user_scoped_tx(pool, user_id) as conn:
            targets = await DevicesRepository(user_id, conn).find_push_targets()
        if not targets:
            logger.info("push.skip user=%s reason=no_targets kind=reminder", user_id)
            return 0
        return await _send_to_targets(
            pool,
            user_id=user_id,
            targets=targets,
            payload=_build_reminder_payload(title=title, message=message, task_id=task_id),
        )
    except Exception:
        logger.exception("push.reminder_dispatch_failed user=%s", user_id)
        return 0
```

```python
def _reminder_card(task: ScheduledTaskRow, *, card_id: str, now: datetime) -> dict[str, Any]:
    payload = task.payload or {}
    raw = payload.get("reminder")
    reminder = raw if isinstance(raw, dict) else {}
    title = str(reminder.get("title") or task.name or "Reminder").strip()
    message = str(reminder.get("message") or title).strip()

        "id": card_id,
        "goalBasket": "life",
        "title": title,
        "whyNow": message,
        "summary": message,
        "primaryAction": {"label": "Done", "type": "confirm"},
        "secondaryActions": [{"label": "Later", "type": "snooze"}],
        "confidence": 1.0,
        "urgency": 0.8,
        "importance": 0.6,
        "sources": ["scheduled-reminder"],
        "agentChain": [],
        "createdAt": now.isoformat(),
        "sourceKind": "scheduled",
        "sourceLabel": title,
    }


async def fire_reminder(
    pool: AsyncConnectionPool,
    task: ScheduledTaskRow,
    run_id: str,
) -> None:
    now = datetime.now(UTC)
    card = _reminder_card(task, card_id=f"card-rem-{uuid.uuid4().hex[:12]}", now=now)
    await persist_pipeline_run_strict(
        pool,
        user_id=task.user_id,
        payload={"runId": run_id, "cards": [card]},
        trigger="reminder",
    )
    logger.info(
        "reminder.card fired user=%d task=%s run=%s title=%r",
        task.user_id, task.id, run_id, card["title"],

    await push_notifier.notify_reminder(
        pool,
        user_id=task.user_id,
        title=card["title"],
        message=card["summary"],
        task_id=task.id,
    )
```

```python
@tool("scheduled_generator_create_once", args_schema=ScheduledGeneratorCreateOnceInput)
async def scheduled_generator_create_once(
    original_request: str,
    timezone: str,
    schedule: dict[str, Any] | ScheduleSpecInput,
    mode: str = "reminder",
    topic: str | None = None,
    reminder: dict[str, Any] | ReminderInput | None = None,
):
    """Create a ONE-OFF scheduled task directly — no preview, no confirmation
    round (one-shots are low-risk and instantly undoable). Use for "5 分钟后
    提醒我喝水" / "remind me at 3pm tomorrow" / "tomorrow 9am give me a news
    digest once". schedule.type MUST be "once". mode="reminder" needs a
    reminder {title, message} you write in the user's language; mode="generate"
    needs a topic. On success returns the task with scheduleLabel — confirm it
    back to the user and tell them they can reply to cancel. On http_400 relay
    the detail (e.g. below minimum interval) plainly."""
    status_code, body = await http_client.post_status(
        f"{_BASE}/once",
        {
            "originalRequest": original_request,
            "timezone": timezone,
            "schedule": _as_dict(schedule),
            "mode": mode,
            "topic": topic,
            "reminder": _as_dict(reminder) if reminder is not None else None,
        },
    )
    if status_code >= 400:
        return {"error": f"http_{status_code}", "detail": body}
    return body
```
