```python
class ReasoningStep(_AgentModel):
    """One user-visible step of the card's "How I reasoned" trace.

    Written by the card-generator LLM in the same structured call that
    writes the card; the evaluator audits each detail against sources[]
    (a step referencing un-cited data is a grounding violation)."""

    label: Literal["observed", "cross_referenced", "pattern_matched", "flagged", "decided"]
    detail: str = Field(min_length=1, max_length=200)

    @field_validator("label", mode="before")
    @classmethod
    def _normalize_label(cls, value: Any) -> Any:
        # LLM 偶发 "Cross-referenced" / "Pattern matched" 变体;归一化到
        # 枚举形,避免一个 label 大小写让整张卡验证失败重试。
        if isinstance(value, str):
            return value.strip().lower().replace("-", "_").replace(" ", "_")
        return value
```

```python
# Maps the first token of a citation prefix (text before " id=") to the
# user-facing app slug stamped onto card["sourceApps"]. Must stay in sync
# with _citation_for above — that template is the only producer of the
# citation strings this parses.
_CITATION_TOKEN_TO_APP = {
    "signal": "device",
    "memory": "memory",
    "gmail": "gmail",
    "gcal": "gcal",
    "world": "web",
    "profile": "profile",
    "goal": "goal",
    "preference": "preference",
}

_CITATION_PREFIX_RE = re.compile(r"^\s*([a-z0-9_-]+)(?:\s+[a-z0-9_-]+)*\s+id=", re.IGNORECASE)


def derive_source_apps(sources: list[str]) -> list[str]:
    """Deterministic user-facing app slugs for a card's citation list.

    Unknown / garbled citations are dropped rather than guessed — the
    audit string itself must never leak to the UI (the old behaviour the
    frontend is moving away from)."""
    apps: list[str] = []
    for citation in sources:
        match = _CITATION_PREFIX_RE.match(citation)
        if not match:
            continue
        token = match.group(1).lower()
        app = _CITATION_TOKEN_TO_APP.get(token, token)
        if app not in apps:
            apps.append(app)
    return apps
```

```ts
const CITATION_TOKEN_TO_APP: Record<string, string> = {
  signal: "device",
  memory: "memory",
  gmail: "gmail",
  gcal: "gcal",
  world: "web",
  profile: "profile",
  goal: "goal",
  preference: "preference",
};

const CITATION_PREFIX_RE = /^\s*([a-z0-9_-]+)(?:\s+[a-z0-9_-]+)*\s+id=/i;

/** Mirror of the backend's derive_source_apps (generator.py) for cards
 *  persisted before the backend stamped sourceApps. Unknown shapes are
 *  dropped so raw audit strings never reach the UI. */
export function deriveSourceAppsFromCitations(sources: string[]): string[] {
  const apps: string[] = [];
  for (const citation of sources) {
    const match = CITATION_PREFIX_RE.exec(citation);
    if (!match) continue;
    const token = match[1].toLowerCase();
    const app = CITATION_TOKEN_TO_APP[token] ?? token;
    if (!apps.includes(app)) apps.push(app);
  }
  return apps;
}

function normalizeReasoningSteps(value: unknown): ReasoningStep[] {
  if (!Array.isArray(value)) return [];
  const steps: ReasoningStep[] = [];
  for (const item of value) {
    if (!isRecord(item)) continue;
    const label = typeof item.label === "string" ? item.label.trim() : "";
    const detail = typeof item.detail === "string" ? item.detail.trim() : "";
    if (!label || !detail) continue;
    steps.push({ label, detail });
  }
  return steps;
}
```
