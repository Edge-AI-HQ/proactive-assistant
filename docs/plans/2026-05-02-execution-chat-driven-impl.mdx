```typescript
function buildFirstMessage(
  card: ChatCardContext | undefined,
  rawCardContext: string | undefined,
): string {
  if (!card && !rawCardContext) return '';
  if (!card) {
    return 'The user opened an intent card. Greet briefly and suggest what you can do based on the card context.';
  }

  const { primaryAction, urgency = 0, title, sources } = card;

  // ── source-kind hint(spec §Tier 政策 / §选定方案 §5)──
  // 让 agent 在第一时间走对应 Tier 1/2 路径,而不是临场决策。
  // sources 字段格式 `<source_kind>:[<metadata>] '<quoted>'`
  const sourceKinds = (sources ?? [])
    .map((s) => s.split(':')[0])
    .filter(Boolean);
  const uniqueKinds = Array.from(new Set(sourceKinds));
  const sourceHint = uniqueKinds.length > 0
    ? ` Sources: ${uniqueKinds.join(', ')}. Use the matching pin Tier 1/2 path first (e.g. \`pin raw gmail\` for gmail, \`pin raw gcal\` for gcal, \`pin memory\` for memory:topic, \`pin world\` for world).`
    : '';

  if (urgency >= 0.7 && primaryAction.type === 'confirm') {
    return `The user opened "${title}" and wants to proceed. Execute: "${primaryAction.label}". If anything is ambiguous, ask one question.${sourceHint}`;
  }
  if (primaryAction.type === 'confirm') {
    return `The user opened "${title}". Suggested action: "${primaryAction.label}". Confirm details briefly, then execute.${sourceHint}`;
  }
  return `The user opened "${title}". Help them with: ${card.summary}${sourceHint}`;
}
```

- [ ] **Step 3:在 `routes/chat.ts` 实现 `shouldInterceptBrowserFirstCall`**

```typescript
// ── Tier 政策轻硬拦截(spec §轻硬拦截)──────────────────────────────────
// 当 agent 第一次 tool call 是 browser open 且本次 run 还没有任何 pin raw /
// pin personal exec 调用时,拦截一次,inject Tier 政策提醒。
//
// 边角:用户最近一条消息显式提到 browser/网页/网站/browse 等关键词时直接放行
// (spec §风险与测试 列项;避免误伤 user-directed 流程)。

const USER_BROWSER_KEYWORDS = /browser|网页|网站|browse|navigate|打开.*url|open.*http/i;

export interface InterceptInput {
  toolName: string;
  toolArgs: unknown;
  priorToolCallsInRun: Array<{ name: string; args: unknown }>;
  userMessage: string;
}

export interface InterceptResult {
  intercept: boolean;
  reminder?: string;
}

export function shouldInterceptBrowserFirstCall(input: InterceptInput): InterceptResult {
  if (input.toolName !== 'browser') return { intercept: false };
  const args = input.toolArgs as { action?: string } | null;
  if (!args || args.action !== 'open') return { intercept: false };

  // User 显式提到 browser → 放行
  if (USER_BROWSER_KEYWORDS.test(input.userMessage)) return { intercept: false };

  // Prior 已有 browser open(说明这是 retry 第二次)→ 放行
  for (const prior of input.priorToolCallsInRun) {
    if (prior.name === 'browser') {
      const a = prior.args as { action?: string } | null;
      if (a?.action === 'open') return { intercept: false };
    }
    if (prior.name === 'exec') {
      const a = prior.args as { command?: string } | null;
      const cmd = typeof a?.command === 'string' ? a.command : '';
      if (/\bpin\s+raw\b/.test(cmd) || /\bpin\s+personal\s+exec\b/.test(cmd)) {
        return { intercept: false };
      }
    }
  }

  return {
    intercept: true,
    reminder: [
      '[Tier policy reminder]',
      'Browser is the LAST-RESORT tool. Try Tier 1 first:',
      '  - Reading Gmail/GCal/Slack/Notion data → `pin raw <toolkit>` or `pin personal exec --tool <READ_TOOL>`',
      '  - Reading public articles → `pin web fetch --url <url>`',
      '  - Side effects → `pin personal exec --tool <WRITE_TOOL> --allow-side-effect=true`',
      'Reconsider: which Tier 1-3 tool fits this task? If browser truly is the only path',
      '(captcha / 2FA / no API), retry the same browser call and it will go through.',
    ].join('\n'),
  };
}
```

在 `handleChat()` 的 `onAgentEvent` 闭包里,`tool` stream + `phase === 'start'` 之前调 hook。如果 `intercept = true`,**不**写 `chat:tool:start` 帧,而是:

具体集成方法取决于 OpenClaw subagent runtime 提供的 API。如果不确定,先用最简版:**inject reminder 走"chat:tool:result"**(content = reminder text),让 agent 自己 LLM-loop 决策下一步。这种实现的妥协是 reminder 进了 transcript(用户能在 history 里看到它),不是隐式 system 提醒——这点 acceptable 给 1-C,Plan 2 时可以重做。

```typescript
if (phase === 'start') {
  const args = evt.data.args;
  // ── Tier 政策轻硬拦截(spec §轻硬拦截)──
  const interceptCheck = shouldInterceptBrowserFirstCall({
    toolName: name,
    toolArgs: args,
    priorToolCallsInRun: priorToolCalls,  // closure 内累积的数组
    userMessage: body.message ?? '',
  });
  if (interceptCheck.intercept) {
    ctx.api.logger.info(`[chat-stream] intercepted browser-first-call toolCallId=${toolCallId}`);
    // 把 reminder 当作 "tool result" 返给 agent,让它自己重决策
    write({
      type: 'chat:tool:result',
      toolCallId,
      name,
      isError: false,
      meta: interceptCheck.reminder,
      timestamp: Date.now(),
    });
    // 关键:把这次 call 计入 prior,下次 retry browser open 不再拦
    priorToolCalls.push({ name, args });
    // ⚠ TODO:实际让 OpenClaw subagent runtime 把 reminder 当系统消息回灌给 agent。
    //   minimal 版只 emit SSE → agent 自己 transcript 里看不到——需用 runtime API
    //   inject。具体待 1-C 实施时确认 OpenClaw subagent.injectMessage 是否存在。
    return;
  }
  // ... 原有 tool:start 转发逻辑 ...
  priorToolCalls.push({ name, args });  // 正常 tool call 也累积
  // ...
}
```

### 风险

- **OpenClaw runtime 没有 mid-run inject API**:如果实地查证 runtime 不支持 mid-run system inject,fallback 到 `chat:tool:result` 把 reminder 当"工具失败/警告结果"返给 agent。Agent LLM 看到一个失败的工具结果会自动 reconsider,这种妥协对 LLM 决策仍然有效。Plan 2 重 UI 时可以把这个 reminder 用更原生的"系统提醒"渲染。
- **拦截影响 SSE streaming**:hook 同步执行,纯 JS 检查 + 字符串匹配,不阻塞;不影响其他 `chat:delta` / `chat:tool:*` 流。
- **`priorToolCalls` 数组只在单次 SSE run 闭包内累积,不跨 run**:意味着用户再发一条新消息开启新 run 时数组为空,如果用户**明确说**"刚才 browser 没成功,再试试" → agent 第一调 browser open 又会被拦。但因为 `userMessage` 含 "browser" 关键词,USER_BROWSER_KEYWORDS 正则会放行 —— 这种 case 边界 ok。
- **Edge case:`browser` tool 第一次就是 `snapshot`(没 open 过)**:正常 agent 不会这样 —— snapshot 必须先 open。如果出现,hook 不拦(action !== 'open'),`browser` 工具实现会自己报错。

5. 用户接手做 §6 个手工场景

### 风险

- **本地不能完整验 Tier 3 副作用闭环**(需要真实 Composio account + 真发邮件 / 真建日程)。本地只验到"agent 起草 + 等用户 OK"环节。完整闭环留 dev 部署后或在 prod 用户机器实地验。
- **场景 3 起草环节的语义判定**:agent 起草后用户回 OK,agent 是否真发取决于本地 Composio account 接入状态;如果账号未连,会失败 —— 这是预期,不算 Plan 1 失败。
- **场景 5 拦截 hook 实际行为**:取决于 §1-C §Step 4 选用的注入方式(SSE tool-result vs runtime inject)。SSE tool-result 路径下 reminder 会出现在 transcript 里,用户能看到;如果不希望用户看到,这是 Plan 2 的 UX 改造范围。
- **Rebase 风险**:如果 v2 这两天有人合了别的 commit 导致 rebase 冲突,优先用代码语义判断(不要无脑 accept HEAD 或 incoming);如果冲突在 1-A reset 涉及的文件,小心 v2 的 commit 不要被覆盖回去。
